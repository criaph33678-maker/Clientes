let clientes = JSON.parse(localStorage.getItem("flextv_clientes")) || []; let escondeContagem = JSON.parse(localStorage.getItem("flextv_ocultar")) || false; let idClienteEmEdicao = null; function salvarNoStorage() { localStorage.setItem("flextv_clientes", JSON.stringify(clientes)); localStorage.setItem("flextv_ocultar", JSON.stringify(escondeContagem)); } function renderizarApp() { const container = document.getElementById("lista-clientes"); const elTotal = document.getElementById("total-clientes"); const btnEye = document.getElementById("btn-toggle-eye"); container.innerHTML = ""; // MODIFICAÇÃO 3: Filtra clientes NÃO arquivados const clientesAtivos = clientes.filter(c => !c.arquivado); // MODIFICAÇÃO 3: Exibe o contador ignorando os arquivados if (escondeContagem) { elTotal.innerText = "***"; btnEye.innerText = "🙈"; } else { elTotal.innerText = clientesAtivos.length; btnEye.innerText = "👁️"; } // MODIFICAÇÃO 1: Ordenação por data de vencimento próxima // Clientes marcados como PAGO com vencimento próximo sobem para o TOPO em VERDE clientesAtivos.sort((a, b) => new Date(a.dataVencimento) - new Date(b.dataVencimento)); if (clientesAtivos.length === 0) { container.innerHTML = '

Nenhum cliente ativo.

'; } else { clientesAtivos.forEach(cliente => { const card = document.createElement("div"); // MODIFICAÇÃO 1: Fica verde quando o cliente tiver status PAGO card.className = `cliente-card ${cliente.pago ? 'pago' : ''}`; card.innerHTML = `

${cliente.nome}

Login: ${cliente.login || 'Não informado'}

Cadastro: ${formatarData(cliente.dataCadastro)}

Vencimento: ${formatarData(cliente.dataVencimento)}

${cliente.pago ? 'PAGO' : ''}
`; container.appendChild(card); }); } } // Botão de alternar olho document.getElementById("btn-toggle-eye").addEventListener("click", () => { escondeContagem = !escondeContagem; salvarNoStorage(); renderizarApp(); }); function formatarData(dataIso) { if (!dataIso) return ""; const partes = dataIso.split("-"); return `${partes[2]}/${partes[1]}/${partes[0]}`; } // Cadastrar Novo Cliente document.getElementById("btn-cadastrar").addEventListener("click", () => { const nome = document.getElementById("novo-nome").value; const login = document.getElementById("novo-login").value; const vencimento = document.getElementById("novo-vencimento").value; if (!nome || !vencimento) { alert("Informe ao menos o nome e a data de vencimento!"); return; } const hoje = new Date().toISOString().split("T")[0]; const novo = { id: Date.now(), nome: nome, login: login, dataCadastro: hoje, // Data fixada na criação dataVencimento: vencimento, pago: false, arquivado: false }; clientes.push(novo); salvarNoStorage(); renderizarApp(); document.getElementById("novo-nome").value = ""; document.getElementById("novo-login").value = ""; document.getElementById("novo-vencimento").value = ""; }); // MODIFICAÇÃO 1: Pede a data para estender ao clicar em PAGO function abrirModalPago(id) { idClienteEmEdicao = id; const cliente = clientes.find(c => c.id === id); document.getElementById("modal-cliente-info").innerText = `Cliente: ${cliente.nome}`; document.getElementById("modal-pago").classList.remove("hidden"); } function fecharModal() { document.getElementById("modal-pago").classList.add("hidden"); document.getElementById("nova-data-vencimento").value = ""; idClienteEmEdicao = null; } // MODIFICAÇÃO 1: Estende a data do vencimento mantendo a data inicial salva e marca verde document.getElementById("btn-confirmar-pago").addEventListener("click", () => { const novaData = document.getElementById("nova-data-vencimento").value; if (!novaData) { alert("Selecione a nova data para prolongar!"); return; } const cliente = clientes.find(c => c.id === idClienteEmEdicao); if (cliente) { cliente.dataVencimento = novaData; // Atualiza a data estendida cliente.pago = true; // Fica verde e reorganiza na lista salvarNoStorage(); } fecharModal(); renderizarApp(); }); document.getElementById("btn-cancelar-modal").addEventListener("click", fecharModal); // MODIFICAÇÃO 3: Arquiva o cliente (Removendo-o da contagem e lista ativa) function arquivarCliente(id) { const cliente = clientes.find(c => c.id === id); if (cliente) { cliente.arquivado = true; salvarNoStorage(); renderizarApp(); } } renderizarApp();
